-- Create table (primary key removed, as employees have multiple quarters)
CREATE TABLE emp_performance (
    emp_id           NUMBER,
    emp_name         VARCHAR2(50),
    department       VARCHAR2(30),
    salary           NUMBER(10,2),
    performance_qtr  VARCHAR2(10),
    score            NUMBER(5,2)
);

-- Analytic function: show previous score and change
SELECT
    emp_name,
    department,
    performance_qtr,
    score,
    LAG(score, 1) OVER (
        PARTITION BY emp_name
        ORDER BY performance_qtr
    ) AS prev_score,
    score - LAG(score, 1) OVER (
        PARTITION BY emp_name
        ORDER BY performance_qtr
    ) AS change_in_score
FROM emp_performance
ORDER BY emp_name, performance_qtr;
